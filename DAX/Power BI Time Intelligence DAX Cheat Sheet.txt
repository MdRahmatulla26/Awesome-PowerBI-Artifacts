Power BI Time Intelligence DAX Cheat Sheet

# Date Functions

  Function   Description                     Example  
                                            
  TODAY()    Returns today's date             TODAY()   
  NOW()      Returns current date and time    NOW()   
  DATE()     Creates a date                   DATE(2026,7,1)   
  YEAR()     Extract year                     YEAR(Date[Date])   
  MONTH()    Extract month                    MONTH(Date[Date])   
  DAY()      Extract day                      DAY(Date[Date])   
  WEEKDAY()  Day of week                      WEEKDAY(Date[Date])   
  WEEKNUM()  Week number                      WEEKNUM(Date[Date])

# Running Total

DAX
Running Total =
CALCULATE(
    [Total Sales],
    FILTER(
        ALL('Date'),
        'Date'[Date] <= MAX('Date'[Date])
    )
)
   
//////////////////////////////////////////////////////  

# Same Period Last Year

DAX
Sales SPLY =
CALCULATE(
    [Total Sales],
    SAMEPERIODLASTYEAR('Date'[Date])
)
   
//////////////////////////////////////////////////////

# Year Over Year (YoY)

DAX
YoY Growth =
DIVIDE(
    [Total Sales] [Sales Previous Year],
    [Sales Previous Year]
)
   
# Parallel Period

DAX
Sales Parallel Year =
CALCULATE(
    [Total Sales],
    PARALLELPERIOD(
        'Date'[Date],
         1,
        YEAR
    )
)

# Dates Between

DAX
Sales Between Dates =
CALCULATE(
    [Total Sales],
    DATESBETWEEN(
        'Date'[Date],
        DATE(2026,1,1),
        DATE(2026,12,31)
    )
)

# Dates In Period

DAX
Rolling 12 Months =
CALCULATE(
    [Total Sales],
    DATESINPERIOD(
        'Date'[Date],
        MAX('Date'[Date]),
         12,
        MONTH
    )
)
